A2. AnalizarFrase

        SI((anterior = 'n' )  ∧ (c= 'b')) ENTONCES
            ESCRIBIR("la frase tiene errores ortograficos")
        SINO 
            ESCRIBIR("La frase no tiene errores ortograficos")
        FIN_SI
        LEER(c)
    FIN_MIENTRAS
